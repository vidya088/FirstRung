rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    function signedIn() { return request.auth != null; }
    function hasProfile() { return signedIn() && exists(/databases/$(database)/documents/users/$(request.auth.uid)); }
    function myRole() { return get(/databases/$(database)/documents/users/$(request.auth.uid)).data.role; }
    function isAdmin() { return hasProfile() && myRole() == 'admin'; }
    function isProvider() { return hasProfile() && myRole() == 'provider'; }

    // Profiles: you can read/edit your own; nobody can make themselves admin.
    match /users/{uid} {
      allow read: if signedIn() && (request.auth.uid == uid || isAdmin());
      allow create: if signedIn() && request.auth.uid == uid && request.resource.data.role in ['student', 'provider'];
      allow update: if signedIn() && ((request.auth.uid == uid && request.resource.data.role == resource.data.role) || isAdmin());
    }

    // Listings: public can read approved ones; providers create/edit their own as 'pending'; only admin approves.
    match /opportunities/{id} {
      allow read: if resource.data.status == 'approved' || (signedIn() && (resource.data.ownerId == request.auth.uid || isAdmin()));
      allow create: if isAdmin() || (isProvider() && request.resource.data.ownerId == request.auth.uid && request.resource.data.status == 'pending');
      allow update: if isAdmin() || (isProvider() && resource.data.ownerId == request.auth.uid && request.resource.data.ownerId == request.auth.uid && request.resource.data.status == 'pending');
      allow delete: if isAdmin() || (isProvider() && resource.data.ownerId == request.auth.uid);
    }

    // Saved listings and applications: private to each user.
    match /saved/{uid} {
      allow read, write: if signedIn() && request.auth.uid == uid;
    }
    match /applications/{uid}/items/{oppId} {
      allow read, write: if signedIn() && request.auth.uid == uid;
    }
  }
}
